Usage guide & code sample

Wrap the table in .k-table-wrap for responsive overflow; use .k-table--striped or .k-table--hover to add readability. Use data-k-table data-k-pagination="true" data-k-page-size="5" with .k-table__pagination to activate built-in responsive pagination with search and sort support.

<!-- Basic Responsive Table -->
<div class="k-table-wrap">
  <table class="k-table k-table--striped k-table--hover">
    <thead><tr><th>Name</th><th>Status</th></tr></thead>
    <tbody><tr><td>Core API</td><td>Healthy</td></tr></tbody>
  </table>
</div>

<!-- Table with Search, Sort & Pagination -->
<div class="k-table-wrap" data-k-table data-k-searchable="true" data-k-sortable="true" data-k-pagination="true" data-k-page-size="5">
  <!-- Optional search toolbar -->
  <div class="k-table__toolbar">
    <label class="k-table__search" for="search-input">
      <span class="k-table__search-icon"><i data-kaui-icon="search"></i></span>
      <input id="search-input" class="k-input k-table__search-input" data-k-table-search type="search" placeholder="Search...">
    </label>
  </div>

  <table class="k-table k-table--striped k-table--hover">
    <thead>
      <tr><th scope="col">Project</th><th scope="col">Owner</th><th scope="col">Status</th></tr>
    </thead>
    <tbody>
      <tr data-k-table-no-results hidden><td colspan="3" class="k-table__empty">No results found.</td></tr>
      <tr><th scope="row">KAUI</th><td>Dang Long</td><td>Active</td></tr>
      <tr><th scope="row">Design Tokens</th><td>Minh Tran</td><td>Review</td></tr>
    </tbody>
  </table>

  <!-- Pagination bar -->
  <div class="k-table__pagination">
    <div class="k-table__pagination-info">
      Showing <strong data-k-table-pagination-start>1</strong> to <strong data-k-table-pagination-end>5</strong> of <strong data-k-table-pagination-total>20</strong> entries
    </div>
    <div class="k-table__pagination-controls">
      <label class="k-table__pagination-size">
        Rows per page:
        <select class="k-select" data-k-table-page-size>
          <option value="5" selected>5</option>
          <option value="10">10</option>
          <option value="25">25</option>
        </select>
      </label>
      <nav class="k-pagination k-pagination--sm k-table__pagination-nav" data-k-table-pagination-nav aria-label="Table pagination"></nav>
    </div>
  </div>
</div>
