Usage guide & code sample

Wrap controls in .k-form-group, pair labels with controls, and use data-k-input-format for automatic masking & interactive pickers.

<!-- 1. Standard text input with label & hint -->
<div class="k-form-group">
  <label for="email" class="k-form-label">Email</label>
  <input id="email" class="k-input" type="email" placeholder="name@example.com">
  <span class="k-form-hint">We never share your email.</span>
</div>

<!-- 2. Input with icon prefix -->
<div class="k-input-wrapper">
  <i data-kaui-icon="mail" class="k-input__icon k-input__icon--left"></i>
  <input class="k-input" type="email" placeholder="user@company.com">
</div>

<!-- 3. Password input with native eye toggle -->
<div class="k-input-wrapper">
  <input type="password" id="user-pwd" class="k-input" placeholder="Password">
  <button type="button" class="k-input__password-toggle" data-k-password-toggle="user-pwd" aria-label="Toggle password"></button>
</div>

<!-- 4. Date Picker (yyyy-MM-dd) -->
<input type="text" class="k-input" data-k-input-format="date" placeholder="yyyy-MM-dd">

<!-- 5. Date & Time Picker with time selector (yyyy-MM-dd hh:ii:ss) -->
<input type="text" class="k-input" data-k-input-format="datetime" placeholder="yyyy-MM-dd hh:ii:ss">

<!-- 6. Time Only with time picker (hh:ii:ss) -->
<input type="text" class="k-input" data-k-input-format="time" placeholder="hh:ii:ss">

<!-- 7. Month Picker (yyyy-MM) -->
<input type="text" class="k-input" data-k-input-format="month" placeholder="yyyy-MM">

<!-- 8. Date Range Validation (start <= end) -->
<div class="k-input-range" data-k-input-range="date">
  <input class="k-input" type="text" data-k-input-format="date" placeholder="Start date">
  <span class="k-input-range__separator">to</span>
  <input class="k-input" type="text" data-k-input-format="date" placeholder="End date">
</div>

<!-- 9. Month Range Validation (start <= end) -->
<div class="k-input-range" data-k-input-range="month">
  <input class="k-input" type="text" data-k-input-format="month" placeholder="Start month">
  <span class="k-input-range__separator">to</span>
  <input class="k-input" type="text" data-k-input-format="month" placeholder="End month">
</div>

<!-- 10. Amount / Money auto-formatted (1.250.000,00) -->
<input type="text" class="k-input" data-k-input-format="amount" placeholder="xxx.xxx.xxx,xx">

<!-- 11. Phone numbers (10 digits, 11 digits, Global) -->
<input type="text" class="k-input" data-k-input-format="phone-10" placeholder="0912-345-678">
<input type="text" class="k-input" data-k-input-format="phone-11" placeholder="02435-678-901">
<input type="text" class="k-input" data-k-input-format="phone-global" placeholder="+84 945777323">
