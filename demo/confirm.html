Confirm Component (.k-confirm)

Focused confirmation dialog with centered status icon, message, and explicit two-way decision actions.

Usage guide & code sample

Add is-open to the confirm container to show it. It features a centered icon, title, message, and separated action buttons.

<div class="k-confirm is-open" role="dialog" aria-modal="true" data-k-backdrop="static">
  <div class="k-confirm__backdrop"></div>
  <div class="k-confirm__dialog">
    <div class="k-confirm__body">
      <div class="k-confirm__icon"><i data-kaui-icon="help"></i></div>
      <h3 class="k-confirm__title">Save changes?</h3>
      <p class="k-confirm__message">Your changes will be saved to this record.</p>
    </div>
    <div class="k-confirm__footer">
      <button type="button" class="k-btn k-btn--outline">Cancel</button>
      <button type="button" class="k-btn k-btn--primary">Confirm</button>
    </div>
  </div>
</div>

Script Callback Guide (Confirm & Cancel Events)

You can listen to button events directly, or wrap the Confirm component in a reusable Promise-based helper function for clean async/await workflows.

Method 1: Direct DOM Event Listeners

// Reference elements
const confirmDialog = document.getElementById('my-confirm');
const btnAccept = document.getElementById('my-confirm-accept');
const btnCancel = document.getElementById('my-confirm-cancel');

// When user clicks Confirm / Accept
btnAccept.addEventListener('click', function() {
  // 1. Run action logic (e.g. call API, persist changes, delete records)
  console.log('Action confirmed!');

  // 2. Close the dialog
  confirmDialog.classList.remove('is-open');
});

// When user clicks Cancel / Dismiss
btnCancel.addEventListener('click', function() {
  // 1. Run cancellation logic
  console.log('Action cancelled.');

  // 2. Close the dialog
  confirmDialog.classList.remove('is-open');
});

Method 2: Promise-based Utility (Reusable async/await Helper)

/**
 * Display a Confirm dialog and resolve user choice via Promise<boolean>
 * @param {string} selector - CSS selector of the confirm dialog (e.g. '#demo-confirm')
 * @returns {Promise<boolean>} true if confirmed, false if cancelled or dismissed with Escape
 */
function showConfirm(selector) {
  return new Promise((resolve) => {
    const dialog = document.querySelector(selector);
    if (!dialog) return resolve(false);

    const acceptBtn = dialog.querySelector('.k-btn--primary, .k-btn--warning, .k-btn--danger');
    const cancelBtn = dialog.querySelector('[data-k-confirm-dismiss], .k-btn--outline, #demo-confirm-cancel');

    function cleanup(result) {
      dialog.classList.remove('is-open');
      if (acceptBtn) acceptBtn.removeEventListener('click', onAccept);
      if (cancelBtn) cancelBtn.removeEventListener('click', onCancel);
      document.removeEventListener('keydown', onKeyDown);
      resolve(result);
    }

    function onAccept() { cleanup(true); }
    function onCancel() { cleanup(false); }
    function onKeyDown(e) {
      if (e.key === 'Escape') cleanup(false);
      if (e.key === 'Enter') cleanup(true);
    }

    if (acceptBtn) acceptBtn.addEventListener('click', onAccept);
    if (cancelBtn) cancelBtn.addEventListener('click', onCancel);
    document.addEventListener('keydown', onKeyDown);

    // Open dialog
    dialog.classList.add('is-open');
  });
}

// === Real-world usage in application code ===
async function handleDeleteItem(itemId) {
  const confirmed = await showConfirm('#demo-confirm-danger');
  if (confirmed) {
    // User clicked Confirm / Delete -> execute deletion
    await apiDelete(itemId);
    alert('Item deleted successfully!');
  } else {
    // User cancelled -> abort
    console.log('Deletion aborted.');
  }
}
